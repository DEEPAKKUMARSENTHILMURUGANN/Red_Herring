    // Render Sidebar Chats List
    function renderChatsList() {
      const listContainer = document.getElementById('chats-list');
      const query = document.getElementById('search-input').value.toLowerCase().trim();

      listContainer.innerHTML = '';

      const filtered = CHATS_DATA.filter(chat => {
        if (currentFilter === 'unread' && chat.unread === 0) return false;
        if (currentFilter === 'groups' && !chat.isGroup) return false;

        if (query) {
          const nameMatch = chat.name.toLowerCase().includes(query);
          const msgMatch = chat.messages.some(m => m.text && m.text.toLowerCase().includes(query));
          return nameMatch || msgMatch;
        }

        return true;
      });

      if (filtered.length === 0) {
        listContainer.innerHTML = `
          <div class="p-8 text-center text-[#54656f] dark:text-[#8696a0]">

    // Interrogation Reply Handler (Outbound bubble + 1.5s auto-reply)
    function sendInterrogationMessage() {
      const input = document.getElementById('message-input');
      const text = input.value.trim();
      if (!text) return;

      const chat = CHATS_DATA.find(c => c.id === currentChatId);
      if (!chat) return;

      const now = new Date();
      const timeStr = now.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });

      // Append Outbound Interrogation Bubble
      chat.messages.push({
        sender: 'You',
        text: text,
        time: timeStr,
        isMe: true,
        status: 'read'
      });

      chat.lastTime = timeStr;
      renderMessages(chat);
      renderChatsList();

      input.value = '';

      // Show typing indicator bar
      const typingBar = document.getElementById('typing-indicator-bar');
      document.getElementById('typing-name').innerText = chat.name;
      typingBar.classList.remove('hidden');

      // Delayed Auto-Reply (1.5s)
      setTimeout(() => {
        typingBar.classList.add('hidden');
        triggerAutoReply(chat);
      }, 1500);
    }

    function triggerAutoReply(chat) {
      if (!chat.autoReplies || chat.autoReplies.length === 0) return;

      // Pick next reply sequentially
      const replyText = chat.autoReplies.shift();
      chat.autoReplies.push(replyText); // cycle back

      const now = new Date();
      const timeStr = now.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });

      chat.messages.push({
        sender: chat.name,
        text: replyText,
        time: timeStr,
        isMe: false
      });

      chat.lastTime = timeStr;

      if (chat.id === currentChatId) {
        renderMessages(chat);
      }
      renderChatsList();
    }
